% ---------- Which margin are we in? ----------
% marginnote calls \raggedleftmarginnote when it \llap's the note, i.e. when the
% note lands in the *left* margin, and \raggedrightmarginnote otherwise. Both
% keep marginnote's own alignment -- ragged left in a left margin, ragged right
% in a right one -- so that margin material always hugs the text block instead
% of the paper edge, and they additionally record which side we are on.
% \ctmarginragged is what the margin blocks below use, since issuing a bare
% \raggedright inside the note would undo that choice on verso pages.
\newif\ifctmarginleft
\renewcommand*{\raggedleftmarginnote}{\ctmarginlefttrue\raggedleft}
\renewcommand*{\raggedrightmarginnote}{\ctmarginleftfalse\raggedright}
\newcommand{\ctmarginragged}{\ifctmarginleft\raggedleft\else\raggedright\fi}

% The rule that opens a margin block. It is set on its own line: \par before it
% kills the indent, \par after closes the (0.9pt tall) line box before the skip.
% \makebox is needed rather than a bare \hfill because \raggedright already
% puts infinite glue at the right, which would share the space with it.
\newlength{\ctmarginrulewidth}
\setlength{\ctmarginrulewidth}{1.6em}
\newcommand{\ctmarginrule}{%
  \par\noindent
  \ifctmarginleft
    \makebox[\linewidth][r]{\rule{\ctmarginrulewidth}{0.9pt}}%
  \else
    \makebox[\linewidth][l]{\rule{\ctmarginrulewidth}{0.9pt}}%
  \fi
  \par\vskip2.5pt}

% ---------- Theorem-like blocks: head in the margin ----------
% The label ("Definition 1.14" + its optional title) is set in the outer
% margin, the one already used by \sidecite. The body carries no frame and no
% background: only two hairlines, above and below, so that the start and the
% end of the block are unambiguous without boxing it in.
% nobreak stops a block from being split across two pages, which mdframed did
% badly whenever the body held a numbered equation (frame closed mid-page,
% rest of the page left blank). \raggedbottom keeps the resulting gap clean,
% and this replaces the \newpage that had to be added by hand.
% Each block is set on a very light wash of its own colour rather than being
% ruled or framed: it marks the start and the end unambiguously, survives a
% page break, and needs no left/right decision (the earlier dashed rule had to
% guess which side the outer margin was on, and got it wrong often enough).
% \ctblockbg is redefined globally by \ctmarginhead before mdframed draws.

% \marginnote is silently dropped when issued from thmtools' headformat (the
% head is typeset in a box that is measured but never shipped out), so the
% label is emitted from the post-head hook instead, where it survives.
% \thmt@thmname holds the environment name and \thmt@optarg its optional
% title; #1 is the environment, used to reach its counter via \the<env>
% (unnumbered environments such as remark simply have none).
% The margin holds two very different things: these labels and the \sidecite
% entries, and colour alone did not tell them apart (it vanishes in greyscale
% and on a printed copy). Two non-chromatic signals are added, both carried by
% the label only: a short rule above it, and a sans-serif uppercase name set
% against the serif of a citation. The optional title stays serif italic, so
% the discriminating line is the first one.
% The label line must not hyphenate ("PROPOSI-TION"); letterspacing is kept low
% (\textls[30], not the [80] of \spacedlowsmallcaps) so that the widest name
% still fits the 6em margin. The title keeps hyphenation on, otherwise long
% words ("CONDITIONNEMENT") overflow the margin.
\newcommand{\ctmarginhead}[1]{%
  \gdef\ctblockbg{ct#1!8}%
  \leavevmode
  \marginnote{%
    \ctmarginragged\hbadness=10000
    \color{ct#1}%
    \ctmarginrule
    {\sffamily\bfseries\fontsize{7.2pt}{8.6pt}\selectfont
     \hyphenpenalty=10000\exhyphenpenalty=10000
     \textls[30]{\MakeTextUppercase{\thmt@thmname}}%
     \@ifundefined{the#1}{}{~\csname the#1\endcsname}\par}%
    \ifx\thmt@optarg\@empty\else
      \vspace{2.5pt}\normalfont\fontsize{7.5pt}{8.5pt}\selectfont\itshape\thmt@optarg
    \fi\par}%
  \ignorespaces}
\makeatother

% babel-french makes `:' an active character (thin space before the colon).
% babel protects \ref, \label and \cite from this via its \@safe@actives
% switch, but it knows nothing about cleveref: \cref{sec:foo} then expands the
% active `:' inside a \csname ... \endcsname and the compilation errors out.
% We therefore wrap the cleveref user commands in that same switch. Removing
% this block requires either dropping the colons in all label names or turning
% off the french shorthand, which would lose the spacing before `:' everywhere.
\makeatletter
\@ifundefined{@safe@activestrue}{}{%
  % NB: cleveref declares these commands with \DeclareRobustCommand. They are
  % used inside \caption, i.e. in a moving argument written to the .lof, so the
  % redefinitions below MUST stay robust as well -- \renewcommand would make
  % them fragile and break the .lof/.aux writing.
  % The originals must be grabbed as cleveref's internal `\cref ' (with the
  % trailing space) and not as `\cref': the latter is only the \protect shell,
  % which would point back at the new definition and loop forever.
  \protected\def\ct@safeactives#1{\@safe@activestrue#1\@safe@activesfalse}%
  \expandafter\let\expandafter\ct@orig@cref\csname cref \endcsname
  \expandafter\let\expandafter\ct@orig@Cref\csname Cref \endcsname
  \expandafter\let\expandafter\ct@orig@crefrange\csname crefrange \endcsname
  \expandafter\let\expandafter\ct@orig@Crefrange\csname Crefrange \endcsname
  \expandafter\let\expandafter\ct@orig@cpageref\csname cpageref \endcsname
  \expandafter\let\expandafter\ct@orig@Cpageref\csname Cpageref \endcsname
  \DeclareRobustCommand*{\cref}[1]{\ct@safeactives{\ct@orig@cref{#1}}}%
  \DeclareRobustCommand*{\Cref}[1]{\ct@safeactives{\ct@orig@Cref{#1}}}%
  \DeclareRobustCommand*{\crefrange}[2]{\ct@safeactives{\ct@orig@crefrange{#1}{#2}}}%
  \DeclareRobustCommand*{\Crefrange}[2]{\ct@safeactives{\ct@orig@Crefrange{#1}{#2}}}%
  \DeclareRobustCommand*{\cpageref}[1]{\ct@safeactives{\ct@orig@cpageref{#1}}}%
  \DeclareRobustCommand*{\Cpageref}[1]{\ct@safeactives{\ct@orig@Cpageref{#1}}}%
}
\makeatother

% biblatex hands \mkbibnamefamily the macro \namepartfamily, not the name it
% holds, so the name has to be expanded to a string before it can be compared

    \noexpand\cv@markfamily@i{#1}}\cv@tmp}

% \cvcite{JOE24,ICML24} -- cites my own work by its mnemonic key rather than by

\DeclareCiteCommand{\cvcite}
  {\usebibmacro{prenote}}
  {\bibhyperref{\mkbibbrackets{\textsc{\MakeLowercase{\thefield{entrykey}}}}}}

  {\usebibmacro{postnote}}

% cvpublist -- the classified publication list. The entries are written out one
% by one rather than filtered out of the .bbl, because the bibtex8 backend sorts
% once, globally, and cannot sort a single list by descending date the way this
% chapter needs. Ordering by hand also keeps a category in the order I want to
% argue it, not the order a sorting scheme happens to produce.
%
% The label is the mnemonic key, so a reader who meets \cvcite{JOE24} in a later
% chapter finds the entry here without an intermediate lookup.
\newcommand{\cv@pubsetup}{%
  % \mkbibnamefamily is declared robust by biblatex, so it must be redefined
  % robustly too; \renewcommand would silently lose the markup. The URL is
  % dropped because HAL is announced once above the list, and repeating it on
  % every entry buries the reference under its own metadata.
  \renewrobustcmd*{\mkbibnamefamily}[1]{\cv@markfamily{##1}}%
  \renewbibmacro*{url+urldate}{}}

% \cvfullcite -- \fullcite is a cite command, so it truncates the author list at
% maxcitenames; a publication list has to name every co-author. The limit cannot
% be raised from inside the list (the options and hooks that would do it are
% preamble-only), so the raise is baked into a cite command of its own.
\DeclareCiteCommand{\cvfullcite}
  {\defcounter{maxnames}{99}%
   \defcounter{minnames}{99}%
   \usebibmacro{prenote}}
  {\usedriver
     {\DeclareNameAlias{sortname}{default}}
     {\thefield{entrytype}}}
  {\multicitedelim}
  {\usebibmacro{postnote}}

% avec un retournement sur le bord long, la colonne de gauche du recto tombe
% en face de la colonne de droite du verso.

% ---------------------------------------------------------------- geometrie
% Tout est place par rapport au coin de la FEUILLE, et le bloc des trois
% marque-pages est centre sur elle dans les deux directions. C'est la condition
% pour que les colonnes du recto et celles du verso se superposent une fois la
% feuille retournee, quel que soit le bord de retournement.
%
%   largeur du bloc = 2*\bmstep + \bmw = 2*6.5 + 6.2 = 19.2 cm
%   marges laterales = (21 - 19.2)/2 = 0.9 cm de chaque cote
%   marges haut/bas  = (29.7 - 22)/2 = 3.85 cm
%
% Toute modification de \bmw, \bmh ou \bmstep impose de recalculer \bmleft
% et \bmtop.
\newlength{\bmw}\setlength{\bmw}{6.2cm}   % largeur d'un marque-page
\newlength{\bmh}\setlength{\bmh}{22cm}    % hauteur d'un marque-page
\newcommand{\bmstep}{6.5}                 % pas entre deux colonnes, en cm
\newcommand{\bmleft}{0.9cm}               % (21 - 19.2)/2
\newcommand{\bmtop}{-3.85cm}              % -(29.7 - 22)/2

% ---------------------------------------------------------------- contenus
% Bandeau de titre d'un marque-page: #1 couleur, #2 surtitre, #3 titre.
\newcommand{\bmhead}[3]{%
  \begingroup\color{#1}%
  {\footnotesize\textsc{#2}}\\[1pt]
  \rule{\linewidth}{0.8pt}\\[2pt]
  {\normalsize\textbf{#3}}\\[1pt]
  \rule{\linewidth}{0.4pt}
  \endgroup\\[4pt]}

% Sous-titre de groupe a l'interieur d'un marque-page.
\newcommand{\bmgroup}[1]{\\[3pt]{\footnotesize\itshape\color{lctx}#1}\\[1pt]}

% Table a deux colonnes: symbole, signification.
\newcommand{\bmtable}[1]{%
  \begingroup
  \footnotesize
  \setlength{\tabcolsep}{0pt}
  \renewcommand{\arraystretch}{1.22}
  \begin{tabular}{@{}p{1.8cm}@{}p{4.0cm}@{}}#1\end{tabular}
  \endgroup}

% Pied de marque-page.
\newcommand{\bmfoot}[1]{\vfill{\scriptsize\color{lsept}#1}}

% ---------------------------------------------------------------- mise en page
% Un marque-page = une boite de taille fixe, tracee en pointilles pour la coupe.
\newcommand{\bmbox}[2]{% #1 = indice de colonne (0,1,2), #2 = contenu
  \begin{scope}[shift={({#1*\bmstep},0)}]
    \draw[lsept!45, dashed, line width=0.2pt] (0,0) rectangle (\bmw,-\bmh);
    % traits de coupe aux quatre coins
    \foreach \x in {0,\bmw} {
      \draw[lctx!70, line width=0.3pt] (\x,0.25) -- (\x,0.05);
      \draw[lctx!70, line width=0.3pt] (\x,-\bmh-0.05) -- (\x,-\bmh-0.25);
    }
    \node[anchor=north west, inner sep=0pt] at (0.2,-0.2)
      {\begin{minipage}[t][\dimexpr\bmh-0.4cm\relax][t]{\dimexpr\bmw-0.4cm\relax}#2\end{minipage}};
  \end{scope}}

% Une feuille = trois marque-pages cote a cote, cales sur le coin du papier.
\newcommand{\bmsheet}[4]{% #1 = mention de pied de page, #2..#4 = contenus
  \null
  \begin{tikzpicture}[remember picture, overlay]
    \begin{scope}[shift={($(current page.north west)+(\bmleft,\bmtop)$)}]
      \bmbox{0}{#2}\bmbox{1}{#3}\bmbox{2}{#4}
      % mention centree sur la feuille: elle tombe au meme endroit au verso
      \node[anchor=north, inner sep=0pt, font=\tiny, text=lsept]
        at (9.6,-\bmh-0.8) {#1};
    \end{scope}
  \end{tikzpicture}
  \newpage}

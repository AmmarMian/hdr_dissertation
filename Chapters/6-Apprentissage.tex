\begin{itemize}
  \item \textbf{D'où l'on vient.} Le \cref{ch:robust-detection} construit des
    détecteurs à partir d'un modèle spécifié jusqu'au bout: glrt, Rao et Wald
    y sont des \emph{dérivations}. Le modèle y fournit \textbf{deux} choses:
    la réponse, et de quoi la noter (loi sous $\hypothesis_0$, seuil, $\pfa$,
    propriété cfar).
  \item \textbf{Ce chapitre.} La seconde de ces deux fonctions disparaît. Ce
    n'est pas affaire de dimension --- le régime $\Nvar\sim\Nobs$ est un
    \emph{mécanisme}, pas une raison.
  \item \textbf{La raison.} Ce que l'on cherche n'est pas une décision ni un
    paramètre, mais un objet que le modèle ne sait pas noter: un graphe que
    l'on veut \emph{lire}, une partition que l'on veut voir coïncider avec une
    vérité terrain. Aucune vraisemblance ne mesure la lisibilité d'un graphe;
    aucune $\pfa$ ne mesure la qualité d'une partition.
  \item \textbf{La thèse, en une phrase.} \emph{Faire de l'apprentissage sur
    des covariances ne dispense pas du modèle; celui-ci revient sous forme de
    correction.}
  \begin{itemize}
    \item \cref{sec:learning-graphes} --- il revient comme
      \textbf{contrainte}: vraisemblance elliptique pénalisée, contrainte à un
      modèle factoriel de rang faible;
    \item \cref{sec:learning-frechet} --- il revient comme \textbf{correction
      asymptotique}: la moyenne de Fréchet des covariances empiriques est
      biaisée, la théorie des matrices aléatoires fournit le terme correctif.
  \end{itemize}
  \item \textbf{Où cela va.} Le \cref{ch:spdnet}, qui suit, ira au terme: plus
    aucune vraisemblance, une réponse issue de paramètres appris, un critère
    qui sera une erreur de classification. Ce chapitre-ci est le pas
    intermédiaire, celui où le modèle a cessé de juger sans cesser d'estimer.
    \wip{Intro: la fenêtre}{Une phrase à écrire ici: le modèle ne disparaît
    pas non plus au \cref{ch:spdnet}, il y revient par la fenêtre sous la
    forme du choix d'une moyenne --- mais c'est l'affaire de ce chapitre-là.}
  \item \textbf{Ce que le chapitre ne fait pas.} Il ne paraphrase pas les deux
    articles, reproduits p.~\paperpage{ECML23} et \paperpage{ICML24}: il pose
    les modèles, énonce les résultats et renvoie aux articles pour les
    démonstrations, les variantes et le détail des campagnes numériques.
\end{itemize}

\subsection{Deux tâches, un même déplacement}
\label{subsec:learning-taches}

Point de départ commun à tout le mémoire: des observations multivariées, dont
on résume la structure de second ordre par une covariance. Ce qui change ici
est \emph{ce que l'on en fait}.

\begin{definition}[{Apprentissage d'un graphe de dépendances}]
\label{def:learning-tache-graphe}
  Étant données $\Nobs$ observations $\{\x_\iobs\}_{\iobs=1}^{\Nobs}$ de
  $\R^{\Nvar}$, on cherche un graphe non orienté
  $\graph=(\vertexset,\edgeset)$ à $\Nvar$ sommets, un par variable, dont les
  arêtes traduisent les dépendances conditionnelles entre variables.
\end{definition}

\begin{definition}[{Partitionnement d'une collection de covariances}]
\label{def:learning-tache-partition}
  Étant données $\Nmat$ collections d'observations
  $\{\X_\imat\}_{\imat=1}^{\Nmat}$, chacune de $\Nobs$ vecteurs de
  $\R^{\Nvar}$ et de covariance propre $\MCov_\imat$, on cherche une
  affectation $\{\classlabel_\imat\}_{\imat=1}^{\Nmat}$ des collections à
  $\Nclass$ groupes, qui regroupe les covariances semblables.
\end{definition}

\begin{itemize}
  \item en apparence sans rapport: l'un lit la structure interne d'une seule
    covariance, l'autre organise une collection entière;
  \item un trait commun les sépare pourtant tous deux du
    \cref{ch:robust-detection}, et ce trait commande tout ce qui suit.
\end{itemize}

\begin{remark}[{Le critère n'est plus dans le modèle}]
\label{rem:learning-critere}
  \begin{itemize}
    \item dans les deux archétypes du \cref{ch:robust-detection}
      (\cref{def:detection-archetype-cible,def:detection-archetype-egalite}), le
      modèle remplit deux fonctions: il fournit l'estimateur (vraisemblance ou
      $M$-estimation) \emph{et} le critère qui juge la décision (loi sous
      $\hypothesis_0$, seuil, $\pfa$, cfar);
    \item que la dispersion y soit paramètre de nuisance ou paramètre testé n'y
      change rien --- c'est même l'unique point commun des deux archétypes;
    \item ici la seconde fonction tombe: modularité, accord avec une lecture
      experte, accord avec une vérité terrain, aucune n'est une fonction de la
      vraisemblance ni ne s'écrit dans le paramètre du modèle;
    \item l'estimateur continue d'en venir --- mais rien, dans le modèle, ne
      dit plus s'il fait l'affaire.
  \end{itemize}
  \leavevmode
\end{remark}

Conséquence: tant que le modèle certifie sa propre sortie, il suffit de le
résoudre; dès qu'il ne la certifie plus, il faut se demander \emph{où} il
échoue et le corriger là. Reste à dire précisément ce qui échoue.

\subsection{Ce que le régime dimensionnel casse}
\label{subsec:learning-rmt}

% Cette sous-section est le contexte minimal de theorie des matrices
% aleatoires. Elle est ecrite ici plutot qu'en \cref{pt:context} parce que rien
% d'autre dans le memoire ne s'en sert. Elle suit l'ordre de la presentation
% HCERES: regime, Marchenko-Pastur, modele spike, puis la consequence sur les
% distances --- qui est la seule chose dont la \cref{sec:learning-frechet} a
% reellement besoin.

\begin{itemize}
  \item la \cref{sec:context-covariance-regularized-estimation} a signalé la
    dégradation de la scm quand $\Nvar$ approche $\Nobs$, par le
    conditionnement: lecture juste mais grossière --- elle dit que l'estimateur
    devient instable, pas \emph{comment};
  \item la théorie des matrices aléatoires le dit exactement, et c'est ce qui
    permet de \emph{corriger} plutôt que de régulariser à l'aveugle.
\end{itemize}

\begin{definition}[{Régime des matrices aléatoires}]
\label{def:learning-regime-rmt}
  On parle de \textbf{régime des matrices aléatoires} lorsque la dimension et
  le nombre d'observations croissent conjointement,
  \begin{equation}
    \Nvar\to\infty,
    \quad
    \Nobs\to\infty,
    \quad
    \frac{\Nvar}{\Nobs}\to\cratio\in(0,1),
    \label{eq:learning-regime-rmt}
  \end{equation}
  par opposition au \textbf{régime asymptotique classique}, où $\Nvar$ reste
  fixe et $\Nobs\to\infty$. Le réel $\cratio$ est appelé \textbf{rapport de
  concentration}.
\end{definition}

\begin{remark}[{Pourquoi ce régime plutôt que l'autre}]
\label{rem:learning-pourquoi-regime}
  Le régime classique fait de la covariance empirique un estimateur consistant,
  et c'est sur lui que reposent les garanties usuelles. Il décrit mal les
  situations de ce chapitre. Dans le partitionnement hyperspectral de la
  \cref{subsec:learning-hyperspectral}, la covariance d'un pixel est estimée
  sur une fenêtre spatiale de quelques dizaines d'échantillons pour une
  dizaine de bandes après réduction; dans les données \textsc{eeg}, la
  situation est du même ordre. Le rapport $\Nvar/\Nobs$ n'y est pas
  négligeable, et le traiter comme s'il l'était revient à supposer résolu le
  seul problème réellement présent. La \cref{def:learning-regime-rmt} n'est pas
  une abstraction: c'est le régime de travail effectif.
\end{remark}

\begin{property}[{Loi de Mar\v{c}enko-Pastur}]
\label{prop:learning-marchenko-pastur}
  Soit $\X\in\R^{\Nvar\times\Nobs}$ à entrées indépendantes, centrées et de
  variance unité, et $\SCM=\frac{1}{\Nobs}\X\X^\T$ la covariance empirique
  associée, de sorte que $\MCov=\Id{\Nvar}$. Dans le régime
  \cref{eq:learning-regime-rmt}, la distribution empirique des valeurs propres
  de $\SCM$ converge vers la densité \sidecite[1.2cm]{marchenko1967distribution}
  \begin{equation}
    \MPdens(\eig)
    =
    \frac{1}{2\pi\cratio\eig}
    \sqrt{(\eig_+-\eig)(\eig-\eig_-)}\,
    \mathbb{1}_{[\eig_-,\eig_+]}(\eig),
    \qquad
    \eig_\pm=(1\pm\sqrt{\cratio})^2 .
    \label{eq:learning-marchenko-pastur}
  \end{equation}
\end{property}

\begin{remark}[{Ce que cette loi dit vraiment}]
\label{rem:learning-lecture-mp}
  La \cref{fig:learning-marchenko} donne à voir l'énoncé, qui mérite d'être lu
  lentement, car il est plus fort qu'il n'en a l'air. La vraie covariance est l'identité: \emph{toutes} ses valeurs propres
  valent $1$. Celles de son estimateur, elles, s'étalent sur tout l'intervalle
  $[(1-\sqrt{\cratio})^2,(1+\sqrt{\cratio})^2]$, et cet étalement ne disparaît
  pas quand on ajoute des données --- il ne dépend que de $\cratio$. À
  $\cratio=1/2$, le spectre observé couvre $[0{,}09;\,2{,}91]$ là où la vérité
  est un point; à $\cratio=1$, il couvre $[0,4]$. Le biais n'est donc ni du
  bruit ni un artefact de petit échantillon: c'est un phénomène déterministe,
  parfaitement décrit, et donc parfaitement corrigible. C'est ce dernier point
  qui fait toute la différence avec la régularisation par rétrécissement de la
  \cref{sec:context-covariance-regularized-estimation}, laquelle contracte le
  spectre sans savoir de combien.
\end{remark}

Le modèle blanc n'est évidemment pas celui des applications: les covariances
que l'on manipule ont une structure. Le modèle de référence est alors le
\textbf{modèle à pics} (\emph{spiked model}), qui superpose quelques directions
énergétiques à un fond isotrope.

\begin{property}[{Déplacement des pics}]
\label{prop:learning-spiked}
  Soit $\MCov=\sum_{\ivar=1}^{\Nrank}\eig_\ivar\vector{u}_\ivar\vector{u}_\ivar^\T+\Id{\Nvar}$
  avec $\eig_\ivar>1$ et $\Nrank$ fixe. Dans le régime
  \cref{eq:learning-regime-rmt}, les valeurs propres de $\SCM$ associées aux
  pics émergent au-dessus du support de
  \cref{eq:learning-marchenko-pastur} et se placent, asymptotiquement, en
  \sidecite[1.2cm]{baik2006eigenvalues,johnstone2001distribution}
  \begin{equation}
    \widehat{\eig}_\ivar
    \;\simeq\;
    \eig_\ivar\left(1+\frac{\cratio}{\eig_\ivar-1}\right),
    \label{eq:learning-spiked}
  \end{equation}
  pourvu que $\eig_\ivar-1>\sqrt{\cratio}$; en deçà de ce seuil, le pic reste
  noyé dans le fond et n'est plus discernable.
\end{property}

Deux enseignements en découlent, et ce sont eux qui portent toute la
\cref{sec:learning-frechet}. D'abord, le biais est \emph{explicite}: la
\cref{eq:learning-spiked} donne le déplacement en fonction de $\cratio$ et de
la valeur propre vraie, donc on peut l'inverser. Ensuite, ce biais affecte tout
ce qui se calcule à partir du spectre --- et la distance affine invariante de
la \cref{prop:context-riemann-distspd} en fait précisément partie, puisqu'elle
ne dépend que des valeurs propres généralisées du couple.

\begin{property}[{Non-consistance de la distance empirique}]
\label{prop:learning-distance-biaisee}
  Soient $\MCov$ une covariance et $\SCM$ son estimateur empirique sur $\Nobs$
  observations. Dans le régime \cref{eq:learning-regime-rmt}, et pour toute
  matrice de référence $\MCov_0\in\SPD{\Nvar}$,
  \begin{equation}
    \geodist^2(\MCov_0,\SCM)
    \;\overset{\text{\footnotesize/}}{\longrightarrow}\;
    \geodist^2(\MCov_0,\MCov).
    \label{eq:learning-distance-biaisee}
  \end{equation}
  Un estimateur consistant $\distrmt^2(\MCov_0,\SCM)$ de $\geodist^2(\MCov_0,\MCov)$
  peut en revanche être construit à partir des seules quantités observables
  $\cratio$ et $\{\eig_\ivar(\MCov_0^{-1}\SCM)\}$
  \sidecite[1.5cm]{couillet2019distances}.
\end{property}

Voilà le point exact où le modèle cesse de suffire, et voilà aussi la forme que
prend son retour. Il ne s'agit pas de renoncer à la distance riemannienne, ni
de la remplacer par une heuristique: il s'agit d'en estimer la valeur vraie à
partir de données bruitées, en utilisant la connaissance précise que l'on a du
bruit. C'est une correction, au sens le plus littéral.

\subsection{Du graphe à la matrice de précision}
\label{subsec:learning-graphe-precision}

Le pont entre un graphe et une matrice passe par l'inverse de la covariance.

\begin{definition}[{Modèle graphique gaussien}]
\label{def:learning-ggm}
  Soit $\x=[x_1,\dots,x_{\Nvar}]^\T\sim\gaussian(\vector{0},\MCov)$ et
  $\Precision=\inv{\MCov}$ sa \textbf{matrice de précision}. Le champ de Markov
  gaussien associé est le graphe non orienté à $\Nvar$ sommets dans lequel une
  arête relie les sommets $\ivar$ et $\ivarbis$ si et seulement si les
  variables $x_\ivar$ et $x_\ivarbis$ sont conditionnellement dépendantes
  sachant toutes les autres \sidecite[1.0cm]{dempster1972covariance,lauritzen1996graphical}.
\end{definition}

\begin{property}[{Support de la précision et arêtes}]
\label{prop:learning-support}
  Avec les notations de la \cref{def:learning-ggm},
  \begin{equation}
    \mathrm{corr}\!\left(x_\ivar x_\ivarbis \mid x_{\intrange{1}{\Nvar}\setminus\{\ivar,\ivarbis\}}\right)
    =
    -\frac{[\Precision]_{\ivar\ivarbis}}{\sqrt{[\Precision]_{\ivar\ivar}[\Precision]_{\ivarbis\ivarbis}}} .
    \label{eq:learning-corr-partielle}
  \end{equation}
  Une entrée non nulle de $\Precision$ hors diagonale équivaut donc à une arête
  du graphe: \emph{apprendre le graphe, c'est estimer le support de la matrice
  de précision}.
\end{property}

L'approche standard estime $\Precision$ par maximum de vraisemblance
régularisé,
\begin{equation}
  \max_{\Precision\in\SPD{\Nvar}}
  \;
  \log\Det(\Precision)-\trace{\SCM\Precision}-\regparam\,\penfun(\Precision),
  \label{eq:learning-glasso}
\end{equation}
avec $\penfun$ une pénalité de parcimonie et $\regparam\in\R_+$ un paramètre de
régularisation. Le choix $\penfun=\norm{\cdot}_1$ donne \algname{glasso}
\sidecite[0cm]{friedman2008sparse}, référence du domaine.

\subsection{Deux insuffisances}
\label{subsec:learning-insuffisances}

Deux objections à \cref{eq:learning-glasso}, et ce sont elles qui motivent tout
le travail.

\paragraph{L'hypothèse gaussienne.}
\begin{itemize}
  \item la vraisemblance de \cref{eq:learning-glasso} est celle du modèle
    gaussien, dont la fragilité aux queues lourdes est le fil conducteur du
    \cref{ch:context-distributions};
  \item les données pour lesquelles on veut un graphe --- réponses
    catégorielles, séries financières, séries géodésiques --- ne sont pas
    gaussiennes, et une seule observation atypique suffit à déplacer des
    arêtes;
  \item remède déjà employé au \cref{ch:robust-detection}: passer à la famille
    elliptique (\cref{def:context-distributions-elliptical}), de
    log-vraisemblance négative
\end{itemize}
\begin{equation}
  \loss(\MCov)
  \;\propto\;
  \frac{1}{\Nobs}\sum_{\iobs=1}^{\Nobs}
  \rho\!\left(\x_\iobs^\T\inv{\MCov}\x_\iobs\right)
  +\frac{1}{2}\log\Det(\MCov),
  \qquad
  \rho(t)=-\log\densgen(t),
  \label{eq:learning-nll-elliptique}
\end{equation}
qui redonne le cas gaussien pour $\densgen(t)=\exp(-t/2)$
\sidecite[0cm]{vogel2011elliptical}. Générateur retenu dans les expériences: la
$t$ de Student à $\dof=5$ degrés de liberté.

\paragraph{L'absence de structure.}
\begin{itemize}
  \item le régime dimensionnel, sous la forme rencontrée en
    \cref{sec:context-covariance-regularized-estimation}: à
    $\Nobs\simeq\Nvar$ ou $\Nobs<\Nvar$, la scm qui alimente
    \cref{eq:learning-glasso} n'est plus fiable;
  \item la parcimonie seule ne compense pas: il faut réduire le nombre de
    paramètres;
  \item façon canonique de le faire: supposer les données engendrées par un
    modèle linéaire de faible dimension.
\end{itemize}

\begin{definition}[{Modèle factoriel de rang faible}]
\label{def:learning-facteur}
  On dit que $\x$ suit un modèle factoriel de rang $\Nrank$ lorsque
  $\x=\Factormat\vector{s}+\vector{\epsilon}$, avec
  $\Factormat\in\R^{\Nvar\times\Nrank}$ et $\vector{s}$, $\vector{\epsilon}$
  indépendants. La covariance est alors de la forme
  $\MCov=\Lowrank+\Diagpart$, où
  $\Lowrank=\Factormat\,\E[\vector{s}\vector{s}^\T]\Factormat^\T$ appartient à
  l'ensemble $\PSDrank{\Nvar}{\Nrank}$ des matrices semi-définies positives de
  rang $\Nrank$, et $\Diagpart\in\Diagpos{\Nvar}$. On note
  \begin{equation}
    \Factorman{\Nvar}{\Nrank}
    =
    \left\{\MCov=\Lowrank+\Diagpart,\;
    \Lowrank\in\PSDrank{\Nvar}{\Nrank},\;
    \Diagpart\in\Diagpos{\Nvar}\right\}
    \label{eq:learning-factorman}
  \end{equation}
  l'ensemble des matrices « rang faible plus diagonale ».
\end{definition}

\begin{remark}[{Le gain en paramètres}]
\label{rem:learning-gain-parametres}
  \begin{itemize}
    \item la dimension du problème passe de $\Nvar(\Nvar+1)/2$ à
      $\Nvar(\Nrank+1)-\Nrank(\Nrank-3)/2$;
    \item pour $\Nvar=33$, $\Nrank=10$ (jeu \texttt{animals} de la
      \cref{subsec:learning-graphe-resultats}): de $561$ à $328$ paramètres;
    \item gain réel sans être spectaculaire --- et ce n'est pas là l'intérêt
      principal du modèle factoriel. La \cref{prop:learning-composantes} donne
      la vraie raison.
  \end{itemize}
  \leavevmode
\end{remark}

\begin{property}[{Rang et composantes connexes}]
\label{prop:learning-composantes}
  Dans les modèles à contrainte laplacienne, une matrice de précision de rang
  $\Nrank$ correspond à un graphe à $\Nvar-\Nrank$ composantes connexes
  \sidecite[1.5cm]{kumar2020unified}.
\end{property}

\begin{itemize}
  \item cet énoncé transforme une contrainte de calcul en contrainte
    \emph{interprétable}: la structure spectrale imposée pour des raisons
    d'estimation gouverne le nombre de groupes que le graphe fera apparaître;
  \item les méthodes concurrentes exigent que ce nombre soit fixé a priori;
    ici il découle du rang.
\end{itemize}

\subsection{Le problème posé, et pourquoi il est difficile}
\label{subsec:learning-probleme-graphe}

En combinant les deux remèdes, le problème d'apprentissage s'écrit
\begin{equation}
  \min_{\MCov\in\SPD{\Nvar}}
  \;
  \loss(\MCov)+\regparam\,\penfun(\MCov)
  \qquad\text{sous contrainte}\qquad
  \MCov\in\Factorman{\Nvar}{\Nrank},
  \label{eq:learning-probleme-gfm}
\end{equation}
avec $\loss$ la vraisemblance elliptique \cref{eq:learning-nll-elliptique} et,
pour pénalité, une fonction élément par élément appliquée à la
\emph{précision}:
\begin{equation}
  \penfun(\MCov)=\sum_{\ivar\neq\ivarbis}\penaltyelem\!\left([\inv{\MCov}]_{\ivar\ivarbis}\right).
  \label{eq:learning-penalite}
\end{equation}

\begin{remark}[{La difficulté est dans le mélange}]
\label{rem:learning-difficulte}
  \begin{itemize}
    \item aucun ingrédient n'est neuf isolément: modèles graphiques elliptiques
      et modèles factoriels existent;
    \item ce qui résiste est leur \emph{réunion}: la vraisemblance
      \cref{eq:learning-nll-elliptique} et la contrainte
      \cref{eq:learning-factorman} portent sur la \textbf{covariance}, la
      pénalité \cref{eq:learning-penalite} sur son \textbf{inverse};
    \item aucun schéma de descente coordonnée à la \algname{glasso} ne sait
      traiter des termes exprimés dans les deux représentations;
    \item réponse habituelle: fractionnement de variables et \algname{admm},
      qui résout un problème \emph{approché} et ajoute ses propres paramètres
      de réglage.
  \end{itemize}
  \leavevmode
\end{remark}

\subsection{Résolution par optimisation riemannienne}
\label{subsec:learning-resolution-graphe}

\begin{itemize}
  \item la contribution de \fullcite{ECML23} est de traiter
    \cref{eq:learning-probleme-gfm} \textbf{directement}, en tirant parti du
    fait que $\Factorman{\Nvar}{\Nrank}$ est une sous-variété lisse de
    $\SPD{\Nvar}$;
  \item le problème contraint devient un problème \emph{libre} sur une
    variété: la machinerie de la \cref{sec:context-riemann-optimization}
    s'applique sans approximation ni paramètre supplémentaire.
\end{itemize}

Sur $\SPD{\Nvar}$ --- contrainte de rang relâchée --- les trois ingrédients
sont ceux du \cref{ch:context-riemann}: métrique affine invariante
\cref{eq:context-riemann-metriqueaffine}, transport parallèle
\cref{eq:context-riemann-transportspd}, et rétraction du second ordre
\begin{equation}
  \Retr_{\MCov}(\tvec)=\MCov+\tvec+\tfrac{1}{2}\tvec\inv{\MCov}\tvec,
  \label{eq:learning-retraction-spd}
\end{equation}
qui garantit la positivité des itérées. Le gradient riemannien s'obtient du
gradient euclidien par
$\Rgrad f(\MCov)=\MCov\,\symop(\nabla f(\MCov))\,\MCov$.

\begin{proposition}[{Gradient riemannien de la vraisemblance elliptique}]
\label{prop:learning-gradient-nll}
  Le gradient riemannien de $\loss$ en $\MCov\in\SPD{\Nvar}$ vaut
  \begin{equation}
    \Rgrad\loss(\MCov)
    =
    \frac{1}{2}\MCov
    -\frac{1}{2\Nobs}\sum_{\iobs=1}^{\Nobs}
    \weight\!\left(\x_\iobs^\T\inv{\MCov}\x_\iobs\right)\x_\iobs\x_\iobs^\T,
    \qquad
    \weight(t)=-2\frac{\densgen'(t)}{\densgen(t)},
    \label{eq:learning-gradient-nll}
  \end{equation}
  et celui de la pénalité \cref{eq:learning-penalite} a pour entrées
  $[\Rgrad\penfun(\MCov)]_{\ivar\ivarbis}=\penaltyelem'([\inv{\MCov}]_{\ivar\ivarbis})$
  hors diagonale, et $0$ sur la diagonale.
\end{proposition}

\begin{remark}[{La fonction de poids, encore}]
\label{rem:learning-poids}
  \begin{itemize}
    \item $\weight$ de \cref{eq:learning-gradient-nll} est exactement celle de
      la $M$-estimation de la \cref{subsec:context-robust-m-estimation}:
      $\weight(t)=1$ dans le cas gaussien,
      $\weight(t)=(\dof+\Nvar)/(\dof+t)$ pour la $t$ de Student;
    \item la robustesse ne vient d'aucun mécanisme nouveau: c'est la même
      pondération des observations par leur distance de Mahalanobis que celle
      qui produit l'estimateur de Tyler au \cref{ch:robust-detection};
    \item exemple net de ce que le mémoire cherche à montrer: le même objet ---
      ici une fonction de poids --- sert à décider dans un chapitre et à
      apprendre dans un autre.
  \end{itemize}
  \leavevmode
\end{remark}

Pour le modèle factoriel, la géométrie retenue est celle des matrices
semi-définies positives de rang fixe
\sidecite[-0.4cm]{bonnabel2009riemannian,bouchard2021lowrank}, combinée à celle
du cône diagonal. Le détail --- paramétrisation, projections, rétraction sur
$\Factorman{\Nvar}{\Nrank}$ --- est dans l'article; on retient ici son effet
pratique.

\begin{remark}[{Le vrai gain est algorithmique}]
\label{rem:learning-gain-algorithmique}
  \begin{itemize}
    \item une itération ne demande que la \textsc{svd} tronquée du facteur de
      rang faible, là où les méthodes à contrainte laplacienne concurrentes
      demandent la décomposition complète du laplacien ou de l'adjacence à
      chaque itération;
    \item le goulot passe de $\bigO(\Nvar^3)$ à $\bigO(\Nvar\Nrank^2)$;
    \item comme au \cref{ch:robust-detection}, la structure ne sert pas
      seulement à mieux estimer: elle sert aussi à calculer moins.
  \end{itemize}
  \leavevmode
\end{remark}

\section{Corriger la moyenne: matrices aléatoires et moyenne de Fréchet}
\label{sec:learning-frechet}

\begin{contributions}
  Cette section reprend \fullcite{ICML24}, reproduit p.~\paperpage{ICML24}.
\end{contributions}

\subsection{Classer et partitionner avec des moyennes}
\label{subsec:learning-centroides}

Commencer par la tâche, car c'est elle qui fait apparaître l'objet à corriger:
$\Nmat$ collections d'observations $\{\X_\imat\}_{\imat=1}^{\Nmat}$, chacune
résumée par sa covariance empirique
$\SCM_\imat=\frac{1}{\Nobs}\X_\imat\X_\imat^\T$, à classer ou à regrouper. La
méthode de référence sur le cône est d'une simplicité désarmante.

\begin{definition}[{Classification au plus proche centroïde}]
\label{def:learning-centroide}
  Phase d'apprentissage: pour chaque classe $\iclass\in\intrange{1}{\Nclass}$,
  on calcule la moyenne de Fréchet des covariances de cette classe,
  \begin{equation}
    \Fmeanclass{\iclass}
    =
    \argmin_{\MCov\in\SPD{\Nvar}}
    \sum_{\imat\,:\,\classlabel_\imat=\iclass}
    \geodist^2(\MCov,\SCM_\imat).
    \label{eq:learning-centroide-fit}
  \end{equation}
  Phase de décision: une covariance $\SCM$ non étiquetée reçoit l'étiquette
  \begin{equation}
    \classlabel
    =
    \argmin_{\iclass\in\intrange{1}{\Nclass}}
    \;
    \geodist^2(\Fmeanclass{\iclass},\SCM).
    \label{eq:learning-centroide-predict}
  \end{equation}
\end{definition}

Le partitionnement en $\Nclass$ groupes suit le même schéma, à ceci près que
les étiquettes sont inconnues: on alterne une étape d'affectation
\cref{eq:learning-centroide-predict} et une étape de recalcul des centroïdes
\cref{eq:learning-centroide-fit}, exactement comme les $\Nclass$-moyennes
euclidiennes \sidecite[0cm]{barachant2011multiclass,congedo2015approximate}.

\begin{remark}[{Tout repose sur la moyenne}]
\label{rem:learning-tout-repose}
  \begin{itemize}
    \item ces deux algorithmes n'ont qu'un seul ingrédient statistique: la
      moyenne de Fréchet de la \cref{def:context-riemann-frechet}. Le reste est
      de la comparaison de distances;
    \item la qualité de la classification ou de la partition est donc
      entièrement gouvernée par celle de cet estimateur: s'il est biaisé, tout
      l'est;
    \item c'est là que la \cref{prop:learning-distance-biaisee} devient un
      problème pratique et non une curiosité asymptotique.
  \end{itemize}
  \leavevmode
\end{remark}

\subsection{La distance corrigée}
\label{subsec:learning-distance-corrigee}

\begin{remark}[{Une normalisation à signaler}]
\label{rem:learning-normalisation}
  \begin{itemize}
    \item cette section emploie la distance affine invariante
      \emph{normalisée},
      $\geodist^2(\MCov_0,\MCov_1)=\frac{1}{2\Nvar}\norm{\logm(\MCov_0^{-1/2}\MCov_1\MCov_0^{-1/2})}_\frob^2$,
      soit \cref{eq:context-riemann-distspd} divisée par $2\Nvar$;
    \item sans effet sur les minimiseurs, donc sur la
      \cref{def:learning-centroide};
    \item indispensable en revanche pour que la quantité admette une limite
      dans le régime \cref{eq:learning-regime-rmt}, où $\Nvar\to\infty$. Les
      formules qui suivent la supposent en place.
  \end{itemize}
  \leavevmode
\end{remark}

\begin{proposition}[{Distance de Fisher corrigée par la théorie des matrices aléatoires}]
\label{prop:learning-distance-corrigee}
  Soient $\MCov_0\in\SPD{\Nvar}$, $\SCM$ la covariance empirique de $\MCov$ sur
  $\Nobs$ observations, et
  $\eigvec=[\eig_1,\dots,\eig_{\Nvar}]^\T$ les valeurs propres de
  $\MCov_0^{-1}\SCM$. Alors
  \begin{multline}
    \distrmt^2(\MCov_0,\SCM)
    =
    \frac{1}{\Nvar}\sum_{\ivar=1}^{\Nvar}
    \left[\tfrac{1}{2}\log^2\eig_\ivar+\log\eig_\ivar\right]
    \\
    -(\eigvec-\rmtaux)^\T
    \left[\frac{1}{\Nvar}\mat{Q}\ones{\Nvar}+\frac{1-\cratio}{\cratio}\vector{q}\right]
    -\frac{1-\cratio}{2\cratio}\log^2(1-\cratio),
    \label{eq:learning-distance-corrigee}
  \end{multline}
  où $\rmtaux$, $\mat{Q}$ et $\vector{q}$ sont des quantités auxiliaires
  entièrement déterminées par $\cratio$ et $\eigvec$, est un estimateur
  consistant de $\geodist^2(\MCov_0,\MCov)$ dans le régime
  \cref{eq:learning-regime-rmt}
  \sidecite[1.5cm]{couillet2019distances}:
  \begin{equation}
    \distrmt^2(\MCov_0,\SCM)
    \;\xrightarrow[\Nvar/\Nobs\to\cratio]{}\;
    \geodist^2(\MCov_0,\MCov).
    \label{eq:learning-consistance}
  \end{equation}
\end{proposition}

\begin{remark}[{Comment lire cette formule}]
\label{rem:learning-lecture-correction}
  \begin{itemize}
    \item le détail des quantités auxiliaires est dans l'article; ce qui compte
      est la \emph{forme} de \cref{eq:learning-distance-corrigee};
    \item premier terme: la distance empirique naïve, à un décalage près. Tout
      le reste est du correctif;
    \item le correctif ne fait intervenir que des quantités \textbf{observables}
      --- le rapport $\cratio=\Nvar/\Nobs$, que l'on connaît, et le spectre
      $\eigvec$, que l'on mesure. Rien n'exige de connaître $\MCov$;
    \item c'est ce qui rend la correction utilisable: elle transforme une
      connaissance théorique du biais en une opération sur les données;
    \item le correctif s'évanouit quand $\cratio\to0$: la formule contient
      l'ancienne comme cas limite.
  \end{itemize}
  \leavevmode
\end{remark}

\subsection{La moyenne corrigée et son gradient}
\label{subsec:learning-moyenne-corrigee}

L'idée tient en une substitution: remplacer $\geodist^2$ par $\distrmt^2$ dans
la définition même de la moyenne de Fréchet.

\begin{definition}[{Moyenne de Fréchet corrigée}]
\label{def:learning-frechet-rmt}
  Étant données $\Nmat$ covariances empiriques
  $\{\SCM_\imat\}_{\imat=1}^{\Nmat}$, on appelle moyenne de Fréchet corrigée
  tout minimiseur de
  \begin{equation}
    \Fmeanrmt
    \in
    \argmin_{\MCov\in\SPD{\Nvar}}
    \;
    \frac{1}{\Nmat}\sum_{\imat=1}^{\Nmat}
    \distrmt^2(\MCov,\SCM_\imat).
    \label{eq:learning-frechet-rmt}
  \end{equation}
\end{definition}

\begin{remark}[{Une substitution qui coûte cher}]
\label{rem:learning-substitution-couteuse}
  \begin{itemize}
    \item la \cref{def:learning-frechet-rmt} est immédiate à écrire, beaucoup
      moins à résoudre;
    \item la moyenne de Fréchet ordinaire dispose d'une condition d'optimalité
      de forme close \cref{eq:context-riemann-frechetopt}, qui donne
      directement le point fixe usuel;
    \item rien de tel ici: $\distrmt^2$ n'est pas la distance d'une métrique
      riemannienne, c'est un \emph{estimateur} d'une distance;
    \item \cref{eq:learning-frechet-rmt} reste un problème d'optimisation sur
      $\SPD{\Nvar}$, et c'est la voie retenue: descente riemannienne avec la
      rétraction \cref{eq:learning-retraction-spd};
    \item ce qui manque alors est le gradient --- c'est lui la contribution
      technique de l'article.
  \end{itemize}
  \leavevmode
\end{remark}

Le gradient s'obtient en deux temps. D'abord, le coût ne dépend de $\MCov$ qu'à
travers le spectre de $\MCov^{-1/2}\SCM_\imat\MCov^{-1/2}$, ce qui ramène le
calcul sur le cône diagonal.

\begin{proposition}[{Réduction au spectre}]
\label{prop:learning-gradient-reduction}
  Soit $f_\imat(\MCov)=g(\Eigmat_\imat)$ où
  $\MCov^{-1/2}\SCM_\imat\MCov^{-1/2}=\Unit_\imat\Eigmat_\imat\Unit_\imat^\T$
  est une décomposition spectrale. Alors
  \begin{equation}
    \Rgrad f_\imat(\MCov)
    =
    -\MCov^{1/2}\Unit_\imat\,
    \inv{\Eigmat_\imat}\,\Rgrad g(\Eigmat_\imat)\,
    \Unit_\imat^\T\MCov^{1/2},
    \label{eq:learning-gradient-reduction}
  \end{equation}
  où $\Rgrad g(\Eigmat_\imat)$ est le gradient riemannien de $g$ en
  $\Eigmat_\imat$ dans $\Diagpos{\Nvar}$.
\end{proposition}

Reste à calculer $\Rgrad g$, c'est-à-dire à dériver
\cref{eq:learning-distance-corrigee} par rapport aux valeurs propres.
\begin{itemize}
  \item expression longue mais entièrement close, faisant intervenir trois
    matrices $\mat{A}$, $\mat{B}$, $\mat{C}$ dont les entrées sont des
    fonctions explicites des $\eig_\ivar$ et de $\Nobs$; elle est à
    p.~\paperpage{ICML24};
  \item le gradient du coût complet en découle par moyenne,
    $\Rgrad h(\MCov)=\frac{1}{\Nmat}\sum_\imat\Rgrad f_\imat(\MCov)$;
  \item l'\cref{alg:learning-frechet-rmt} s'écrit alors sans difficulté.
\end{itemize}

\begin{algorithm}[htb]
  \caption{Moyenne de Fréchet corrigée par la théorie des matrices aléatoires}
  \label{alg:learning-frechet-rmt}
  \begin{algorithmic}[1]
    \Require $\{\X_\imat\}_{\imat=1}^{\Nmat}$ dans $\R^{\Nvar\times\Nobs}$,
      point initial $\MCov_0$, tolérance $\varepsilon>0$, budget $\Niter$
    \For{$\imat\in\intrange{1}{\Nmat}$}
      \State $\SCM_\imat\gets\frac{1}{\Nobs}\X_\imat\X_\imat^\T$
    \EndFor
    \Repeat
      \State calculer $\Rgrad h(\MCov_\iiter)$ par
        \cref{eq:learning-gradient-reduction}
      \State calculer le pas $t_\iiter$ par recherche linéaire
      \State $\MCov_{\iiter+1}\gets\Retr_{\MCov_\iiter}\!\left(-t_\iiter\Rgrad h(\MCov_\iiter)\right)$
      \State $\iiter\gets\iiter+1$
    \Until{$\geodist^2(\MCov_\iiter,\MCov_{\iiter-1})<\varepsilon$ ou $\iiter>\Niter$}
    \Ensure $\Fmeanrmt=\MCov_\iiter$
  \end{algorithmic}
\end{algorithm}

\begin{remark}[{Dériver à la main, encore}]
\label{rem:learning-deriver-main}
  \begin{itemize}
    \item le \cref{ch:spdnet} reprendra le même geste et le même argument:
      l'objet géométriquement juste n'a pas de forme close, sa minimisation est
      itérative, et le travail consiste à en dériver le gradient explicitement
      plutôt qu'à laisser la différentiation automatique parcourir les itérées;
    \item la raison n'y sera pas la même: là-bas, économiser la mémoire d'un
      graphe de calcul qui stockerait toutes les itérées; ici, plus simplement,
      avoir un algorithme du tout;
    \item le constat, lui, est identique --- un des rares fils qui traverse les
      deux chapitres au niveau technique et non au niveau des idées.
  \end{itemize}
  \leavevmode
\end{remark}

\subsection{Ce que la correction achète}
\label{subsec:learning-mse}

Premier critère, \emph{interne}: à quelle distance de la vraie moyenne se
trouve la moyenne estimée? Erreur quadratique moyenne sur données synthétiques,
où la vérité est connue. C'est le critère qui mesure le \emph{mécanisme} décrit
en \cref{subsec:learning-rmt}.

\begin{figure}[htb]
  \centering
  \input{gfx/generated/learning_frechet_mse/mse/plot.tex}
  \provMseQR[1cm]
  \caption[Erreur d'estimation de la moyenne de Fréchet]{Erreur quadratique
    moyenne de l'estimation de la moyenne de Fréchet, en dimension
    $\Nvar=64$, sur $100$ tirages, la moyenne vraie ayant un conditionnement
    de $100$. Cinq estimateurs: la moyenne de Fréchet des scm, celle des
    covariances rétrécies linéairement (Ledoit-Wolf, \algname{oas}), celle des

    \cref{eq:learning-frechet-rmt}. Les bandes sont les interpercentiles
    $5/95$. À gauche, l'erreur en fonction du nombre d'échantillons $\Nobs$
    par covariance, à $\Nmat=10$ matrices; à droite, en fonction du nombre de
    matrices $\Nmat$, à $\Nobs=128$ échantillons. \provMse}
  \label{fig:learning-mse}
\end{figure}

Les deux panneaux de la \cref{fig:learning-mse} ne disent pas la même chose, et
c'est le second qui porte l'argument.
\textbf{Panneau de gauche --- contre $\Nobs$.}
\begin{itemize}
  \item l'erreur de toutes les méthodes décroît quand $\Nobs$ augmente, et la
    moyenne corrigée reste la plus faible sur tout l'intervalle;
  \item elle gagne $8{,}1$~dB sur la scm à $\Nobs=65$ --- dimension et nombre
    d'échantillons presque égaux --- et l'écart tombe à $2{,}2$~dB à
    $\Nobs=300$;
  \item signature d'un \emph{biais de régime} et non d'une variance: ce que la
    correction rattrape est ce que $\cratio=\Nvar/\Nobs$ mesure, et cela
    s'efface quand $\cratio$ devient négligeable;
  \item ce panneau confirme le mécanisme de la \cref{subsec:learning-rmt}. Il
    ne surprendra personne.
\end{itemize}

\textbf{Panneau de droite --- contre $\Nmat$.} C'est lui qui porte l'argument,
et il est contre-intuitif: on s'attendrait à ce qu'accumuler des matrices
compense le manque d'échantillons.
\begin{itemize}
  \item l'erreur de la scm décroît d'abord puis \emph{plafonne}:
    $12{,}7$~dB à $\Nmat=3$, $8{,}2$~dB à $\Nmat=100$, l'essentiel de la
    baisse acquis dès $\Nmat=20$;
  \item celle de la moyenne corrigée continue de descendre, jusqu'à
    $-4{,}3$~dB;
  \item l'écart entre les deux passe de $1{,}4$~dB à $\Nmat=3$ à $12{,}5$~dB à
    $\Nmat=100$: il ne se referme pas, il \textbf{s'ouvre}.
\end{itemize}
referme pas, il \emph{s'ouvre}.
\begin{remark}[{Pourquoi moyenner ne suffit pas}]
\label{rem:learning-moyenner-ne-suffit-pas}
  \begin{itemize}
    \item moyenner réduit la variance, pas le biais;
    \item les $\Nmat$ covariances empiriques ne sont pas indépendantes dans
      leur erreur: elles partagent le déplacement spectral de la
      \cref{prop:learning-spiked}, qui ne dépend que de $\cratio$ et non du
      tirage. Ce déplacement survit intact à la moyenne;
    \item passé un certain $\Nmat$, la variance a disparu et il ne reste que
      lui: c'est là que les méthodes se séparent, et que la correction devient
      le seul recours;
    \item le régime « beaucoup de matrices mal estimées » est exactement celui
      du partitionnement hyperspectral, où chaque pixel fournit une covariance
      sur une petite fenêtre.
  \end{itemize}
  \leavevmode
\end{remark}

Un mot sur les concurrents, dont la position n'est pas accidentelle.
\begin{itemize}
  \item le rétrécissement linéaire (Ledoit-Wolf, \algname{oas}) est \emph{le
    plus mauvais} des cinq, et son erreur ne décroît presque pas --- ni avec
    $\Nobs$, ni avec $\Nmat$;
  \item le rétrécissement non linéaire fait mieux et suit la scm de près;
  \item différence de \textbf{nature} et non de degré: ces méthodes
    régularisent chaque covariance \emph{avant} de moyenner --- elles corrigent
    un objet dont on ne veut pas --- là où la méthode proposée corrige la
    \emph{distance} que la moyenne minimise;
  \item contracter chaque spectre vers un multiple de l'identité rapproche
    chaque matrice de la cible, mais introduit dans toutes le même biais de
    contraction, que la moyenne conserve là encore.
\end{itemize}
